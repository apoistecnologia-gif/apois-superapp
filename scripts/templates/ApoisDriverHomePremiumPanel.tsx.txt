// APOIS_DRIVER_HOME_PREMIUM_V1
// Interface inicial do Motorista — sem alterar dispatch, GPS, aceite nem recusa.
function IdleBottomPanel({
  online,stats,gpsMessage,canUseTestLocation,onUseTestLocation,
}: {
  driverName:string;online:boolean;disabled:boolean;toggleOnline:()=>void;
  stats:{rides:number;earnings:number};gpsMessage:string|null;loading:boolean;
  canUseTestLocation:boolean;onUseTestLocation:()=>void;
}){
  const navigate=useNavigate();
  const [expanded,setExpanded]=useState(false);
  const [sessionStart,setSessionStart]=useState<number|null>(null);
  const [clock,setClock]=useState(()=>Date.now());
  useEffect(()=>{
    // Mede somente a sessão ativa neste dispositivo, não inventa horas anteriores.
    const key="apois:driver:online-session-start-v1";
    if(!online){
      try{sessionStorage.removeItem(key);}catch{}
      setSessionStart(null);
      return;
    }
    const now=Date.now();
    let started=now;
    try{
      const raw=Number(sessionStorage.getItem(key));
      if(Number.isFinite(raw)&&raw>0&&raw<=now)started=raw;
      else sessionStorage.setItem(key,String(now));
    }catch{}
    setSessionStart(started);
  },[online]);
  useEffect(()=>{
    if(!online)return;
    const id=window.setInterval(()=>setClock(Date.now()),30_000);
    return()=>window.clearInterval(id);
  },[online]);
  const minutes=online && sessionStart!==null
    ? Math.max(0,Math.floor((clock-sessionStart)/60_000)):null;
  const sessionTime=minutes===null?"—":
    minutes>=60 ? Math.floor(minutes/60)+"h "+(minutes%60)+"min":minutes+"min";
  const metrics=[
    {label:"Corridas",value:String(stats.rides),icon:ApoisHomeCar,color:"#E51B23"},
    {label:"Ganhos",value:formatBRL(stats.earnings||0),icon:ApoisHomeWallet,color:"#0BA34B"},
    {label:"Tempo online",value:sessionTime,icon:ApoisHomeClock,color:"#E51B23"},
    {label:"Taxa de aceitação",value:"—",icon:ApoisHomeChart,color:"#E51B23"}
  ];
  return(
    <section data-apois-driver-home-v1="true"
      aria-label="Resumo do motorista"
      className="absolute inset-x-0 z-30 flex min-h-0 flex-col rounded-t-[18px] bg-white text-[#14171B] shadow-[0_-3px_14px_rgba(0,0,0,.10)]"
      style={{bottom:"calc(max(env(safe-area-inset-bottom), 7px) + 58px)",height:expanded?"min(36dvh, 290px)":"116px"}}>
      {!expanded && (gpsMessage || canUseTestLocation) && (
        <div className="absolute inset-x-3 bottom-full mb-2 max-h-[76px] overflow-y-auto rounded-xl border border-red-100 bg-white px-3 py-2 shadow-sm">
          {gpsMessage && <p role="status" className="text-[11px] font-semibold text-[#B70C24]">{gpsMessage}</p>}
          {canUseTestLocation && <button type="button" onClick={onUseTestLocation} className="min-h-8 text-[11px] font-bold text-[#B70C24]">Usar localização de teste no Brasil</button>}
        </div>
      )}
      <div className="flex h-10 shrink-0 items-center justify-between gap-1 px-3">
        <button type="button" aria-expanded={expanded} aria-controls="apois-driver-summary-details" onClick={()=>setExpanded(v=>!v)}
          className="flex min-h-10 min-w-0 items-center gap-1 text-left text-[14px] font-black">
          <h2 className="truncate">Resumo de hoje</h2>
          <ChevronRight size={16} className={expanded?"-rotate-90":"rotate-90"}/>
        </button>
        <button type="button" onClick={()=>navigate({to:"/motorista/ganhos"})}
          className="inline-flex min-h-10 shrink-0 items-center gap-1 text-[11px] font-bold text-[#D50923]">
          Ver ganhos <ChevronRight size={15}/>
        </button>
      </div>
      <div className="grid shrink-0 grid-cols-4 divide-x divide-zinc-100 px-1 pb-1">
        {metrics.map((metric)=>(
          <div key={metric.label} className="flex min-w-0 flex-col items-center gap-0.5 px-0.5 text-center">
            <metric.icon size={16} style={{color:metric.color}}/>
            <strong className="w-full truncate text-[clamp(10px,3.2vw,14px)] font-black tabular-nums leading-tight" title={metric.value}>{metric.value}</strong>
            <span className="text-[9px] font-semibold leading-[10px] text-[#656D77]">{metric.label}</span>
          </div>
        ))}
      </div>
      <p className="shrink-0 px-1 text-center text-[9px] leading-tight text-[#757B84]">
        Tempo desta sessão. Aceitação ainda não disponível.
      </p>
      <div id="apois-driver-summary-details" hidden={!expanded} className="min-h-0 flex-1 space-y-2 overflow-y-auto px-3 pb-2 pt-2 [scrollbar-width:none]">
        <p className="text-xs font-semibold">{online?"Disponível para novas corridas":"Ative seu status para receber corridas"}</p>
        {gpsMessage && <p role="status" className="rounded-lg bg-red-50 p-2 text-xs text-[#B70C24]">{gpsMessage}</p>}
        {canUseTestLocation && <button type="button" onClick={onUseTestLocation} className="min-h-10 w-full rounded-lg border border-dashed border-[#B70C24] text-xs font-bold text-[#B70C24]">Usar localização de teste no Brasil</button>}
        <button type="button" onClick={()=>navigate({to:"/motorista/ganhos"})}
          className="flex min-h-10 w-full items-center gap-2 rounded-xl bg-[#FFF0F2] px-3 text-left text-xs font-bold text-[#C50B25]">
          <ApoisHomeChart size={18} className="shrink-0"/>
          <span className="flex-1">{online?"Siga online e aumente seus ganhos!":"Pronto para voltar a dirigir?"}</span>
          <ChevronRight size={16}/>
        </button>
      </div>
    </section>
  );
}
