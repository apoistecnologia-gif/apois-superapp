
  // APOIS_PASSENGER_QA217_MAP_V1: fit newly calculated itinerary above the
  // bottom sheet. Do not recenter an active driver's live-trip camera.
  useEffect(() => {
    if (status !== "ready" || focusPickup || isTrip || approachMode ||
        validDriverLocation || !origin || !destination ||
        displayedRoutePoints.length < 2) return;
    const routeKey = encodedPolyline || route?.encodedPolyline || "";
    if (!routeKey || routePreviewKeyRef.current === routeKey) return;
    routePreviewKeyRef.current = routeKey;
    const timer = window.setTimeout(() => {
      if (nativeModeRef.current) {
        void mapService.fitRouteBounds(nativeMapIdRef.current, displayedRoutePoints);
        return;
      }
      const maps = (window as any).google?.maps;
      if (!maps?.LatLngBounds || !mapRef.current) return;
      const bounds = new maps.LatLngBounds();
      displayedRoutePoints.forEach((point) => bounds.extend(point));
      const height = mapElementRef.current?.clientHeight ?? 600;
      mapRef.current.fitBounds(bounds, {
        top: 105,
        right: 28,
        bottom: Math.min(350, Math.round(height * 0.47)),
        left: 28,
      });
    }, 350);
    return () => window.clearTimeout(timer);
  }, [status, focusPickup, isTrip, approachMode, validDriverLocation,
      origin, destination, encodedPolyline, route?.encodedPolyline, displayedRoutePoints]);


  // APOIS_SEARCH_CAMERA_BREATH_V1: zoom suave sobre o embarque real.
  // Movimento independente dos motoristas, mas sem deslocar suas coordenadas.
  // Nunca executar durante trajeto aceito, encerrado, ou quando passageiro move o mapa.
  useEffect(() => {
    if (status !== "ready" || !focusPickup || !approachMode || !origin || isStatic) return;
    let disposed = false;
    let timer: number | null = null;
    let phase = 0;
    const cycle = async () => {
      if (disposed) return;
      if (document.visibilityState !== "visible") {
        if (!disposed) timer = window.setTimeout(() => void cycle(), 5500);
        return;
      }
      if (mapService.isCameraMovedByUser()) return;
      const zoom = phase++ % 2 === 0 ? 15.6 : 16;
      const center = pickupCameraCoordinate(
        origin, mapElementRef.current?.clientHeight ?? 480, zoom
      );
      try {
        if (nativeModeRef.current) {
          await mapService.setCamera(nativeMapIdRef.current, { coordinate: center, zoom }, true);
        } else if (mapRef.current) {
          mapRef.current.panTo(center);
          mapRef.current.setZoom(zoom);
        }
      } catch {
        // A câmera nao interrompe a solicitacao caso o Maps rejeite animacao.
      } finally {
        if (!disposed) timer = window.setTimeout(() => void cycle(), 5500);
      }
    };
    timer = window.setTimeout(() => void cycle(), 2200);
    return () => {
      disposed = true;
      if (timer !== null) window.clearTimeout(timer);
    };
  }, [status, focusPickup, approachMode, isStatic, origin?.lat, origin?.lng]);

  // Only real coarse dots from the protected RPC. No pins during an active
  // ride. Blink red on JavaScript and Capacitor-native maps.
  useEffect(() => {
    const points = (nearbyDrivers ?? []).slice(0, 12).filter((point) =>
      Number.isFinite(point.lat) && Number.isFinite(point.lng) &&
      Math.abs(point.lat) <= 90 && Math.abs(point.lng) <= 180);
    nearbyMarkerRefs.current.forEach((marker) => marker.setMap(null));
    nearbyMarkerRefs.current = [];
    if (status !== "ready" || !points.length) {
      if (nativeModeRef.current) {
        void mapService.setNearbyDriverMarkers(nativeMapIdRef.current, []).catch(() => undefined);
      }
      return;
    }
    if (nativeModeRef.current) {
      // One update per confirmed server response. Do not blink/replace markers every 1600ms.
      void mapService.setNearbyDriverMarkers(nativeMapIdRef.current, points, false)
        .catch(() => undefined);
      return () => {
        void mapService.setNearbyDriverMarkers(nativeMapIdRef.current, []).catch(() => undefined);
      };
    }
    const maps = (window as any).google?.maps;
    if (!maps?.Marker) return;
    // APOIS_REALISTIC_DRIVER_CAR_V1: foto-render 3D, sem simbolos desenhados.
    // Localização aproximada e anônima, retornada pela RPC protegida.
    nearbyMarkerRefs.current = points.map(position => new maps.Marker({
      map: mapRef.current,
      position,
      clickable: false,
      optimized: false,
      zIndex: 20,
      title: "Motorista próximo (localização aproximada)",
      icon: {
        url: "assets/markers/apois-nearby-car-3d.webp",
        scaledSize: new maps.Size(54, 39),
        anchor: new maps.Point(27, 19),
      },
    }));
    return () => {
      nearbyMarkerRefs.current.forEach((marker) => marker.setMap(null));
      nearbyMarkerRefs.current = [];
    };
  }, [nearbyDrivers, status]);
